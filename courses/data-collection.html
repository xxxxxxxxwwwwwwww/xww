答案解析

解题思路

1. 使用Pandas的isnull()和sum()方法识别并统计缺失值

2. 根据数据类型选择合适的填充方法：数值型数据使用均值或中位数，分类型数据使用众数

3. 使用fillna()方法填充缺失值

代码示例

import pandas as pd
import numpy as np

# 创建示例数据
data = {
    'A': [1, 2, np.nan, 4, 5],
    'B': [np.nan, 2, 3, 4, np.nan],
    'C': ['x', 'y', 'z', np.nan, 'x']
}
df = pd.DataFrame(data)

# 识别缺失值
print("缺失值统计:")
print(df.isnull().sum())

# 处理缺失值
df['A'] = df['A'].fillna(df['A'].mean())
df['B'] = df['B'].fillna(df['B'].median())
df['C'] = df['C'].fillna(df['C'].mode()[0])

print("\n处理后的数据:")
print(df)

import hashlib

def anonymize_data(data, field):
    """对敏感字段进行脱敏处理"""
    if field in data.columns:
        data[field] = data[field].apply(
            lambda x: hashlib.md5(str(x).encode()).hexdigest()[:8]
        )
    return data

网络爬虫原理

网络爬虫是一种自动获取网页内容的程序，通过模拟浏览器行为来抓取数据。

工作流程：

  1. 从初始URL开始（种子URL）
  2. 发送HTTP请求获取网页内容
  3. 解析网页提取数据和新的URL
  4. 将新URL加入待爬队列
  5. 重复过程直到满足停止条件
import requests
from urllib.parse import urljoin, urlparse

class SimpleCrawler:
    def __init__(self, start_url):
        self.start_url = start_url
        self.visited = set()
        self.to_visit = [start_url]
    
    def fetch_page(self, url):
        """获取页面内容"""
        headers = {'User-Agent': 'Mozilla/5.0'}
        try:
            response = requests.get(url, headers=headers, timeout=10)
            if response.status_code == 200:
                return response.text
        except Exception as e:
            print(f"Error fetching {url}: {e}")
        return None
    
    def extract_links(self, html, base_url):
        """提取页面中的链接"""
        from bs4 import BeautifulSoup
        soup = BeautifulSoup(html, 'html.parser')
        links = []
        for a in soup.find_all('a', href=True):
            full_url = urljoin(base_url, a['href'])
            if urlparse(full_url).netloc == urlparse(base_url).netloc:
                links.append(full_url)
        return links
    
    def crawl(self, max_pages=10):
        """执行爬取"""
        while self.to_visit and len(self.visited) < max_pages:
            url = self.to_visit.pop(0)
            if url in self.visited:
                continue
            
            print(f"Crawling: {url}")
            self.visited.add(url)
            
            html = self.fetch_page(url)
            if html:
                # 提取新链接
                new_links = self.extract_links(html, url)
                for link in new_links:
                    if link not in self.visited:
                        self.to_visit.append(link)
        
        print(f"Crawled {len(self.visited)} pages")

def get_random_headers():
    """生成随机请求头"""
    return {
        'User-Agent': ua.random,
        'Accept': 'text/html,application/xhtml+xml',

def crawl_with_retry(url, max_retries=3):
    """带重试机制的爬取"""
    for i in range(max_retries):
        try:
            headers = get_random_headers()
            # proxy = random.choice(proxies_pool)
            
            response = requests.get(
                url, 
                headers=headers,
                # proxies=proxy,
                timeout=10
            )
            
            if response.status_code == 200:
                return response.text
            elif response.status_code == 429:
                wait_time = 2 ** i + random.random()
                print(f"Rate limited, waiting {wait_time}s")
                time.sleep(wait_time)
            
        except Exception as e:
            print(f"Attempt {i+1} failed: {e}")
            time.sleep(2 ** i)
    
    return None

# 使用示例
def crawl_pages(urls):
    results = []
    for url in urls:
        html = crawl_with_retry(url)
        if html:
            results.append(html)
        # 随机延迟
        time.sleep(random.uniform(1, 3))
    return results

API基本概念

API（Application Programming Interface）是软件系统之间交互的接口，允许不同应用程序之间进行数据交换。

RESTful API调用

RESTful API是一种基于HTTP协议的API设计风格，使用标准HTTP方法操作资源。

import requests
import pandas as pd

class RESTAPIClient:
    def __init__(self, base_url):
        self.base_url = base_url
        self.session = requests.Session()
        self.session.headers.update({
            'Content-Type': 'application/json',
            'Accept': 'application/json'
        })
    
    def get(self, endpoint, params=None):
        """GET请求获取资源"""
        response = self.session.get(
            f"{self.base_url}/{endpoint}",
            params=params
        )
        return self._handle_response(response)
    
    def post(self, endpoint, data=None):
        """POST请求创建资源"""
        response = self.session.post(
            f"{self.base_url}/{endpoint}",
            json=data
        )
        return self._handle_response(response)
    
    def put(self, endpoint, data=None):
        """PUT请求更新资源"""
        response = self.session.put(
            f"{self.base_url}/{endpoint}",
            json=data
        )
        return self._handle_response(response)
    
    def delete(self, endpoint):
        """DELETE请求删除资源"""
        response = self.session.delete(
            f"{self.base_url}/{endpoint}"
        )
        return self._handle_response(response)
    
    def _handle_response(self, response):
        """处理响应"""
        if response.status_code in [200, 201]:
            return response.json()
        else:
            raise Exception(f"API Error: {response.status_code}")

def basic_auth(url, username, password):
    credentials = base64.b64encode(
        f"{username}:{password}".encode()
    ).decode()
    headers = {
        'Authorization': f'Basic {credentials}'
    }
    response = requests.get(url, headers=headers)

def bearer_token_auth(url, token):
    headers = {
        'Authorization': f'Bearer {token}'
    }
    response = requests.get(url, headers=headers)
    return response.json()

# 4. OAuth 2.0流程
class OAuth2Client:
    def __init__(self, client_id, client_secret, token_url):
        self.client_id = client_id
        self.client_secret = client_secret
        self.token_url = token_url
        self.access_token = None
    
    def get_token(self):
        """获取访问令牌"""
        response = requests.post(self.token_url, data={
            'grant_type': 'client_credentials',
            'client_id': self.client_id,
            'client_secret': self.client_secret
        })
        data = response.json()
        self.access_token = data['access_token']
        return self.access_token
    
    def make_request(self, url):
        """使用令牌发起请求"""
        if not self.access_token:
            self.get_token()
        headers = {
            'Authorization': f'Bearer {self.access_token}'
        }
        return requests.get(url, headers=headers).json()

# 5. 签名认证（如阿里云API）
def sign_request(params, secret):
    """生成请求签名"""
    sorted_params = sorted(params.items())
    query_string = '&'.join([f"{k}={v}" for k, v in sorted_params])
    signature = base64.b64encode(
        hmac.new(
            secret.encode(),
            query_string.encode(),
            hashlib.sha256
        ).digest()
    ).decode()
    return signature

def parse_json_data(data, path=None):
    """根据路径提取嵌套JSON数据"""
    if path is None:
        return data
    
    keys = path.split('.')
    result = data
    for key in keys:
        if isinstance(result, dict):
            result = result.get(key)
        elif isinstance(result, list):
            result = [item.get(key) for item in result]
    return result

# JSON转DataFrame

    """将JSON数据转换为DataFrame"""
    if record_path:
        df = pd.json_normalize(data, record_path=record_path)

        df = pd.json_normalize(data)
    return df

# 1. IQR方法
def detect_outliers_iqr(data, column):
    Q1 = data[column].quantile(0.25)
    Q3 = data[column].quantile(0.75)
    IQR = Q3 - Q1
    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR
    
    outliers = data[(data[column] < lower) | (data[column] > upper)]
    return outliers, lower, upper

outliers, lower, upper = detect_outliers_iqr(df, 'value')
print(f"IQR方法 - 异常值数量: {len(outliers)}")
print(f"正常范围: [{lower:.2f}, {upper:.2f}]")

# 2. Z-score方法
def detect_outliers_zscore(data, column, threshold=3):
    z_scores = np.abs(stats.zscore(data[column]))
    outliers = data[z_scores > threshold]
    return outliers

outliers_z = detect_outliers_zscore(df, 'value')

import os
import shutil
import sqlite3
import pandas as pd
from datetime import datetime
import zipfile

# ========== 文件备份 ==========
def backup_file(src, backup_dir='backups'):
    """备份单个文件"""
    if not os.path.exists(backup_dir):
        os.makedirs(backup_dir)
    
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    filename = os.path.basename(src)
    name, ext = os.path.splitext(filename)
    backup_name = f"{name}_{timestamp}{ext}"
    backup_path = os.path.join(backup_dir, backup_name)
    
    shutil.copy2(src, backup_path)
    return backup_path

# ========== 数据库备份 ==========
def backup_sqlite_db(db_path, backup_dir='backups'):

    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    backup_path = os.path.join(backup_dir, f"backup_{timestamp}.db")
    
    # 方法1：直接复制文件
    shutil.copy2(db_path, backup_path)
    
    # 方法2：使用SQLite的备份API
    # source = sqlite3.connect(db_path)
    # dest = sqlite3.connect(backup_path)
    # source.backup(dest)
    # source.close()
    # dest.close()
    
    return backup_path

# ========== 数据导出备份 ==========
def export_data_backup(df, base_name='data', backup_dir='backups'):
    """将数据导出为多种格式备份"""
    if not os.path.exists(backup_dir):
        os.makedirs(backup_dir)
    
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    
    # 导出为多种格式
    df.to_csv(f"{backup_dir}/{base_name}_{timestamp}.csv", index=False)
    df.to_json(f"{backup_dir}/{base_name}_{timestamp}.json", orient='records')
    df.to_pickle(f"{backup_dir}/{base_name}_{timestamp}.pkl")
    
    print(f"备份完成: {backup_dir}/{base_name}_{timestamp}.*")

# ========== 压缩备份 ==========
def create_zip_backup(files, backup_name='backup'):
    """创建压缩备份"""
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    zip_name = f"{backup_name}_{timestamp}.zip"
    
    with zipfile.ZipFile(zip_name, 'w', zipfile.ZIP_DEFLATED) as zf:
        for file in files:
            if os.path.exists(file):
                zf.write(file, os.path.basename(file))
    
    return zip_name

# ========== 恢复数据 ==========
def restore_from_backup(backup_path, restore_path):
    """从备份恢复数据"""
    if backup_path.endswith('.zip'):
        with zipfile.ZipFile(backup_path, 'r') as zf:
            zf.extractall(restore_path)
    else:
        shutil.copy2(backup_path, restore_path)
    
    print(f"数据已恢复到: {restore_path}")

完整的数据采集流程包括数据获取、解析、清洗和存储。

import requests
from bs4 import BeautifulSoup
import pandas as pd
import time
import random
import sqlite3
from datetime import datetime

class DataCollectionPipeline:
    def __init__(self, name):
        self.name = name
        self.data = []
    
    def fetch(self, url, params=None, headers=None):
        """获取网页数据"""
        default_headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'
        }
        headers = headers or default_headers
        
        try:
            response = requests.get(url, params=params, headers=headers, timeout=10)
            if response.status_code == 200:
                return response
        except Exception as e:
            print(f"请求失败: {e}")
        return None
    
    def parse(self, html, parser='html.parser'):
        """解析HTML内容"""
        soup = BeautifulSoup(html, parser)
        return soup
    
    def extract(self, soup, selectors):
        """根据选择器提取数据"""
        results = []
        items = soup.select(selectors['container'])
        
        for item in items:
            data = {}
            for field, selector in selectors['fields'].items():
                element = item.select_one(selector)
                data[field] = element.text.strip() if element else None
            results.append(data)
        
        return results
    
    def clean(self, data):
        """数据清洗"""
        df = pd.DataFrame(data)
        
        # 去除空值
        df = df.dropna(subset=['title'])
        
        # 数据类型转换
        if 'price' in df.columns:
            df['price'] = pd.to_numeric(df['price'], errors='coerce')
        
        # 添加采集时间
        df['collected_at'] = datetime.now()
        
        return df
    
    def save(self, df, table_name, db_path='data.db'):
        """保存到数据库"""
        conn = sqlite3.connect(db_path)
        df.to_sql(table_name, conn, if_exists='append', index=False)
        conn.close()
        print(f"已保存 {len(df)} 条记录到 {table_name}")
    
    def run(self, urls, selectors, delay=(1, 3)):
        """执行完整采集流程"""
        for url in urls:
            print(f"正在采集: {url}")
            
            response = self.fetch(url)
            if response:
                soup = self.parse(response.text)
                data = self.extract(soup, selectors)
                df = self.clean(data)
                self.save(df, 'products')
            
            # 随机延迟
            time.sleep(random.uniform(*delay))

import pandas as pd
import numpy as np
import sqlite3
from datetime import datetime

class DataPreprocessor:
    def __init__(self, df):
        self.df = df.copy()
        self.report = {}
    
    def analyze(self):
        """分析数据质量"""
        self.report['shape'] = self.df.shape
        self.report['missing'] = self.df.isnull().sum().to_dict()
        self.report['duplicates'] = self.df.duplicated().sum()
        self.report['dtypes'] = self.df.dtypes.astype(str).to_dict()
        return self.report
    
    def handle_missing(self, strategy='mean'):
        """处理缺失值"""
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns
        
        for col in numeric_cols:
            if self.df[col].isnull().any():
                if strategy == 'mean':
                    self.df[col].fillna(self.df[col].mean(), inplace=True)
                elif strategy == 'median':
                    self.df[col].fillna(self.df[col].median(), inplace=True)
        
        # 分类列用众数填充
        cat_cols = self.df.select_dtypes(include=['object']).columns
        for col in cat_cols:
            if self.df[col].isnull().any():
                self.df[col].fillna(self.df[col].mode()[0], inplace=True)
        
        return self.df
    
    def remove_duplicates(self):
        """去除重复值"""
        before = len(self.df)
        self.df = self.df.drop_duplicates()
        after = len(self.df)
        print(f"去除重复: {before - after} 条")
        return self.df
    
    def detect_outliers(self, column, method='iqr'):
        """检测异常值"""
        if method == 'iqr':
            Q1 = self.df[column].quantile(0.25)
            Q3 = self.df[column].quantile(0.75)
            IQR = Q3 - Q1
            lower, upper = Q1 - 1.5*IQR, Q3 + 1.5*IQR
            outliers = self.df[(self.df[column] < lower) | 
                              (self.df[column] > upper)]
        return outliers
    
    def transform(self, column, method='standard'):
        """数据转换"""
        from sklearn.preprocessing import StandardScaler, MinMaxScaler
        
        if method == 'standard':
            scaler = StandardScaler()
        elif method == 'minmax':
            scaler = MinMaxScaler()
        
        self.df[f'{column}_{method}'] = scaler.fit_transform(
            self.df[[column]]
        )
        return self.df
    
    def save_to_db(self, table_name, db_path='processed.db'):
        """保存到数据库"""
        conn = sqlite3.connect(db_path)
        self.df.to_sql(table_name, conn, if_exists='replace', index=False)
        conn.close()
        print(f"数据已保存到 {db_path}.{table_name}")
