  • 返回处理后的DataFrame

答案解析

解题思路

1. 使用Pandas的isnull()和sum()方法识别并统计缺失值

2. 根据数据类型选择合适的填充方法：数值型数据使用均值或中位数，分类型数据使用众数

3. 使用fillna()方法填充缺失值

代码示例

import pandas as pd
import numpy as np

# 创建示例数据
data = {
    'A': [1, 2, np.nan, 4, 5],
    'B': [np.nan, 2, 3, 4, np.nan],
    'C': ['x', 'y', 'z', np.nan, 'x']
}
df = pd.DataFrame(data)

# 识别缺失值
print("缺失值统计:")
print(df.isnull().sum())

# 处理缺失值
df['A'] = df['A'].fillna(df['A'].mean())
df['B'] = df['B'].fillna(df['B'].median())
df['C'] = df['C'].fillna(df['C'].mode()[0])

print("\n处理后的数据:")
print(df)

时间序列分析方法

常用方法：移动平均、指数平滑、ARIMA、季节性分解等。选择合适的方法取决于数据特征和预测目标。

import pandas as pd
import numpy as np

# 生成示例销售数据
np.random.seed(42)
dates = pd.date_range(start='2023-01-01', periods=24, freq='M')
trend = np.linspace(100, 150, 24)
seasonality = 20 * np.sin(np.arange(24) * np.pi / 6)
noise = np.random.normal(0, 5, 24)
sales = trend + seasonality + noise

df = pd.DataFrame({'日期': dates, '销售额': sales})

# 简单移动平均
df['MA_3'] = df['销售额'].rolling(window=3).mean()
df['MA_6'] = df['销售额'].rolling(window=6).mean()

# 指数平滑
alpha = 0.3
df['ES'] = df['销售额'].ewm(alpha=alpha).mean()

print("=== 时间序列预测方法比较 ===")
print(df[['日期', '销售额', 'MA_3', 'MA_6', 'ES']].tail(6).round(2))

# 预测下一期
next_ma3 = df['销售额'].tail(3).mean()
next_es = alpha * df['销售额'].iloc[-1] + (1-alpha) * df['ES'].iloc[-1]
print(f"\n下期预测 - 移动平均: {next_ma3:.2f}")
print(f"下期预测 - 指数平滑: {next_es:.2f}")

预测模型构建与评估

评估指标包括：MAE（平均绝对误差）、MAPE（平均绝对百分比误差）、RMSE（均方根误差）等。

import pandas as pd
import numpy as np

# 预测评估指标计算
actual = np.array([120, 135, 142, 128, 155, 168])
forecast = np.array([115, 140, 138, 135, 148, 160])

# 计算各种误差指标
errors = actual - forecast
abs_errors = np.abs(errors)
pct_errors = np.abs(errors / actual) * 100

# MAE - 平均绝对误差
mae = np.mean(abs_errors)

# MAPE - 平均绝对百分比误差
mape = np.mean(pct_errors)

# RMSE - 均方根误差
rmse = np.sqrt(np.mean(errors ** 2))

# MSE - 均方误差
mse = np.mean(errors ** 2)

print("=== 预测模型评估 ===")
print(f"MAE: {mae:.2f}")
print(f"MAPE: {mape:.2f}%")
print(f"RMSE: {rmse:.2f}")
print(f"MSE: {mse:.2f}")

# 预测准确率
accuracy = 100 - mape
print(f"\n预测准确率: {accuracy:.2f}%")

# 评估标准
print("\n--- MAPE评估标准 ---")
print("< 10%: 优秀")
print("10-20%: 良好")
print("20-50%: 一般")
print("> 50%: 需要改进")

需求预测实战案例

综合案例：使用多种方法对产品需求进行预测，并选择最优模型。

import pandas as pd
import numpy as np

# 产品需求预测案例
np.random.seed(42)
months = pd.date_range('2023-01', periods=18, freq='M')
base_demand = 500
trend = np.linspace(0, 100, 18)
seasonal = 50 * np.sin(np.arange(18) * 2 * np.pi / 12)
demand = base_demand + trend + seasonal + np.random.normal(0, 20, 18)

df = pd.DataFrame({'月份': months, '实际需求': demand})

# 方法1: 简单移动平均
df['SMA_预测'] = df['实际需求'].rolling(3).mean().shift(1)

# 方法2: 加权移动平均
weights = [0.5, 0.3, 0.2]
df['WMA_预测'] = df['实际需求'].rolling(3).apply(
    lambda x: sum(x * weights), raw=True).shift(1)

# 方法3: 指数平滑
alpha = 0.4
df['ES_预测'] = df['实际需求'].ewm(alpha=alpha).mean().shift(1)

# 计算各方法MAPE
def calc_mape(actual, forecast):
    mask = ~np.isnan(forecast)
    return np.mean(np.abs((actual[mask] - forecast[mask]) / actual[mask])) * 100

methods = ['SMA_预测', 'WMA_预测', 'ES_预测']
results = []
for method in methods:
    mape = calc_mape(df['实际需求'].values, df[method].values)
    results.append({'方法': method.replace('_预测', ''), 'MAPE': f'{mape:.2f}%'})

print("=== 预测方法比较 ===")
print(pd.DataFrame(results))
print("\n推荐使用MAPE最低的方法进行预测")

库存管理基础

库存类型：原材料、在制品、成品。库存管理目标是在保证服务水平的前提下最小化库存成本。

import pandas as pd

# 库存分类管理 - ABC分析
inventory_items = pd.DataFrame({
    '产品编号': ['P001', 'P002', 'P003', 'P004', 'P005', 'P006', 'P007', 'P008'],
    '年销售额': [500000, 350000, 200000, 100000, 80000, 50000, 30000, 20000],
    '库存数量': [1000, 800, 500, 400, 300, 200, 150, 100],
    '单价': [100, 150, 80, 50, 40, 30, 20, 15]
})

# 按销售额排序
inventory_items = inventory_items.sort_values('年销售额', ascending=False)

# 计算累计占比
inventory_items['销售额占比'] = inventory_items['年销售额'] / inventory_items['年销售额'].sum() * 100
inventory_items['累计占比'] = inventory_items['销售额占比'].cumsum()

# ABC分类
def abc_classify(cum_pct):
    if cum_pct <= 80:
        return 'A'
    elif cum_pct <= 95:
        return 'B'
    else:
        return 'C'

inventory_items['ABC分类'] = inventory_items['累计占比'].apply(abc_classify)

print("=== ABC库存分类分析 ===")
print(inventory_items[['产品编号', '年销售额', '累计占比', 'ABC分类']])

# 分类统计
abc_summary = inventory_items.groupby('ABC分类').agg({
    '产品编号': 'count',
    '年销售额': 'sum'
}).rename(columns={'产品编号': '产品数量'})
print("\n分类汇总:")
print(abc_summary)

库存成本分析

库存成本包括：持有成本（仓储、资金占用、损耗）、订货成本、缺货成本。总成本优化是库存管理的核心。

import pandas as pd
import numpy as np

# 库存成本结构分析
D = 10000  # 年需求量
S = 500    # 每次订货成本
H = 20     # 单位持有成本
C = 100    # 单位采购成本

# EOQ经济订货量计算
eoq = np.sqrt(2 * D * S / H)

# 总成本计算
order_times = D / eoq
annual_order_cost = order_times * S
average_inventory = eoq / 2
annual_holding_cost = average_inventory * H
total_cost = annual_order_cost + annual_holding_cost

print("=== EOQ模型分析 ===")
print(f"年需求量: {D}单位")
print(f"每次订货成本: {S}元")
print(f"单位持有成本: {H}元/年")
print(f"\n经济订货量(EOQ): {eoq:.0f}单位")
print(f"年订货次数: {order_times:.1f}次")
print(f"年订货成本: {annual_order_cost:.0f}元")
print(f"年持有成本: {annual_holding_cost:.0f}元")
print(f"年总库存成本: {total_cost:.0f}元")

# 不同订货量成本比较
Q_values = [200, 400, 500, 600, 800, 1000]
print("\n=== 不同订货量成本比较 ===")
for Q in Q_values:
    tc = (D/Q)*S + (Q/2)*H
    print(f"订货量{Q}: 总成本{tc:.0f}元")

库存优化模型

常用模型：EOQ（经济订货量）、EPQ（经济生产量）、连续盘点模型、定期盘点模型等。

import numpy as np
import pandas as pd

# 多产品库存优化
products = pd.DataFrame({
    '产品': ['A', 'B', 'C', 'D'],
    '年需求': [5000, 3000, 2000, 1000],
    '订货成本': [400, 350, 300, 250],
    '持有成本': [25, 30, 20, 15],
    '单价': [80, 100, 60, 40]
})

# 计算各产品EOQ
products['EOQ'] = np.sqrt(2 * products['年需求'] * products['订货成本'] / products['持有成本'])
products['年订货次数'] = products['年需求'] / products['EOQ']
products['平均库存'] = products['EOQ'] / 2
products['年订货成本'] = products['年订货次数'] * products['订货成本']
products['年持有成本'] = products['平均库存'] * products['持有成本']
products['年总成本'] = products['年订货成本'] + products['年持有成本']

print("=== 多产品库存优化 ===")
print(products[['产品', '年需求', 'EOQ', '年订货次数', '年总成本']].round(0))

# 库存投资分析
products['库存投资'] = products['平均库存'] * products['单价']
total_investment = products['库存投资'].sum()
print(f"\n总库存投资: {total_investment:.0f}元")

# 服务水平与安全库存关系
service_levels = [0.85, 0.90, 0.95, 0.99]
z_values = [1.04, 1.28, 1.65, 2.33]  # 正态分布Z值
sigma_d = 50  # 需求标准差
L = 7/365  # 提前期（年）

print("\n=== 服务水平与安全库存 ===")
for sl, z in zip(service_levels, z_values):
    ss = z * sigma_d * np.sqrt(L * 365)
    print(f"服务水平{sl*100:.0f}%: 安全库存{ss:.0f}单位")

安全库存计算

安全库存用于应对需求和供应的不确定性。计算需考虑服务水平、需求波动、提前期波动等因素。

import numpy as np
import pandas as pd
from scipy import stats

# 安全库存计算参数
avg_demand = 100  # 平均日需求
std_demand = 20   # 需求标准差
lead_time = 7     # 提前期（天）
std_lead_time = 2 # 提前期标准差
service_level = 0.95  # 目标服务水平

# Z值（服务水平对应）
z = stats.norm.ppf(service_level)

# 方法1: 仅考虑需求不确定性
ss_demand = z * std_demand * np.sqrt(lead_time)

# 方法2: 同时考虑需求和提前期不确定性
ss_combined = z * np.sqrt(lead_time * std_demand**2 + 
                          avg_demand**2 * std_lead_time**2)

# 再订货点计算
rop = avg_demand * lead_time + ss_combined

print("=== 安全库存计算 ===")
print(f"平均日需求: {avg_demand}单位")
print(f"需求标准差: {std_demand}")
print(f"提前期: {lead_time}天")
print(f"目标服务水平: {service_level*100:.0f}%")
print(f"Z值: {z:.2f}")
print(f"\n安全库存(仅需求不确定性): {ss_demand:.0f}单位")
print(f"安全库存(综合不确定性): {ss_combined:.0f}单位")
print(f"再订货点: {rop:.0f}单位")

# 不同服务水平对比
print("\n=== 不同服务水平的安全库存 ===")
for sl in [0.90, 0.95, 0.99]:
    z_val = stats.norm.ppf(sl)
    ss = z_val * np.sqrt(lead_time * std_demand**2 + avg_demand**2 * std_lead_time**2)
    print(f"服务水平{sl*100:.0f}%: 安全库存{ss:.0f}单位, 再订货点{avg_demand*lead_time+ss:.0f}单位")

运输路径优化

车辆路径问题(VRP)是物流优化的核心问题，目标是最小化运输距离或成本，同时满足容量和时间约束。

import numpy as np
import pandas as pd

# 简化的车辆路径问题
# 客户位置和需求
customers = pd.DataFrame({
    '客户': ['仓库', 'C1', 'C2', 'C3', 'C4', 'C5', 'C6'],
    'X坐标': [0, 10, 15, 20, 35, 30, 5],
    'Y坐标': [0, 20, 25, 10, 15, 5, 30],
    '需求量': [0, 20, 15, 25, 30, 18, 22]
})

# 计算距离矩阵
def calc_distance(p1, p2):
    return np.sqrt((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2)

n = len(customers)
dist_matrix = np.zeros((n, n))
for i in range(n):
    for j in range(n):
        dist_matrix[i,j] = calc_distance(
            (customers.iloc[i]['X坐标'], customers.iloc[i]['Y坐标']),
            (customers.iloc[j]['X坐标'], customers.iloc[j]['Y坐标'])
        )

print("=== 距离矩阵（部分） ===")
print(f"仓库到各客户距离: {dist_matrix[0, 1:].round(1)}")

# 最近邻启发式算法
def nearest_neighbor(dist_matrix, demands, capacity):
    n = len(demands)
    visited = [0]  # 从仓库出发
    route = [0]
    remaining = list(range(1, n))
    current_load = 0
    
    while remaining:
        current = route[-1]
        nearest = min(remaining, key=lambda x: dist_matrix[current][x])
        if current_load + demands[nearest] <= capacity:
            route.append(nearest)
            current_load += demands[nearest]
            remaining.remove(nearest)
        else:
            route.append(0)  # 返回仓库
            current_load = 0
    
    route.append(0)  # 最终返回仓库
    return route

vehicle_capacity = 60
route = nearest_neighbor(dist_matrix, customers['需求量'].values, vehicle_capacity)
print(f"\n推荐路线: {' -> '.join([customers.iloc[i]['客户'] for i in route])}")
