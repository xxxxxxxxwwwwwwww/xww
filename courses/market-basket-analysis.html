from collections import Counter

transactions = [
    ['面包', '牛奶', '啤酒'],
    ['面包', '尿布', '啤酒', '鸡蛋'],
    ['牛奶', '尿布', '鸡蛋', '可乐'],

]

all_items = [item for trans in transactions for item in trans]
unique_items = set(all_items)
print(f"商品种类数: {len(unique_items)}")
print(f"商品列表: {unique_items}")

avg_items = len(all_items) / len(transactions)
print(f"平均每笔交易商品数: {avg_items:.2f}")

item_counts = Counter(all_items)
print("\n商品出现频率:")
for item, count in item_counts.most_common():
    print(f"  {item}: {count}次 ({count/len(transactions)*100:.1f}%)")

模块1：购物篮分析基础（8学时）

购物篮分析概述

购物篮分析（Market Basket Analysis）是零售业中一种重要的数据分析技术，通过分析顾客购买行为中的商品关联，发现商品之间的关联规律，为商品陈列、促销策略、交叉销售等提供数据支持。

📊 核心目标

发现商品之间的关联关系，帮助零售商优化商品布局和营销策略。

🎯 应用场景

商品推荐、货架优化、促销设计、库存管理、用户画像。

关联规则指标

关联规则用于描述商品之间的关联关系，主要有三个核心指标：支持度、置信度和提升度。

📈 支持度 (Support)

Support(A→B) = P(A∪B) = 包含A和B的交易数 / 总交易数

表示A和B同时出现的概率。

📊 置信度 (Confidence)

Confidence(A→B) = P(B|A) = P(A∪B) / P(A)

表示购买A的顾客中购买B的比例。

🎯 提升度 (Lift)

Lift(A→B) = Confidence(A→B) / P(B)

表示关联规则的强度，>1表示正相关。

模块2：Apriori算法（10学时）

Apriori先验原理

Apriori算法是最经典的频繁项集挖掘算法，其核心思想基于先验性质：如果一个项集是频繁的，那么它的所有子集也一定是频繁的。反过来，如果一个项集是非频繁的，那么它的所有超集也一定是非频繁的。

🔑 先验性质

频繁项集的所有非空子集也必须是频繁的。

✂️ 剪枝策略

利用先验性质进行剪枝，减少候选项集数量。

def has_infrequent_subset(candidate, frequent_itemsets):

关联规则生成

从频繁项集中生成关联规则，计算每条规则的支持度、置信度和提升度，筛选出满足阈值的强关联规则。

from itertools import combinations

def generate_association_rules(frequent_itemset, transactions, min_confidence):
    """生成关联规则"""
    rules = []
    items = list(frequent_itemset)
    
    for i in range(1, len(items)):
        for antecedent in combinations(items, i):
            antecedent = set(antecedent)
            consequent = frequent_itemset - antecedent
            
            support_XY = count_support(frequent_itemset, transactions)
            support_X = count_support(antecedent, transactions)
            
            if support_X > 0:
                confidence = support_XY / support_X
                if confidence >= min_confidence:
                    rules.append({
                        'antecedent': antecedent,
                        'consequent': consequent,
                        'confidence': confidence
                    })
    
    return rules

# 示例：频繁项集 {面包, 牛奶, 啤酒}

recommendations = {
    '面包': ['牛奶', '啤酒', '鸡蛋'],
    '牛奶': ['面包', '鸡蛋'],
    '啤酒': ['面包', '尿布'],
}

def recommend_products(cart_items, recommendations):
    suggestions = set()
    for item in cart_items:
        if item in recommendations:
            suggestions.update(recommendations[item])
    return suggestions - set(cart_items)

cart = ['面包']
suggestions = recommend_products(cart, recommendations)
print(f"购买 {cart} 的用户还可能购买:")
print(f"  推荐: {suggestions}")
